const SPREADSHEET_ID = '151XT0NLPb7fljwUOe_PP3dhD_-dPhBmA_jrffxGZWjQ'; function doPost(e) { try { const data = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.openById(SPREADSHEET_ID); // 1. UPDATE STATUS (Kelola Anggota) if (data.action === 'updateStatus') { const sheet = ss.getSheetByName('Sheet1'); const values = sheet.getDataRange().getValues(); for (let i = 1; i < values.length; i++) { // Cocokkan berdasarkan NAMA (Kolom A / index 0) if (String(values[i][0]).trim() === String(data.nama).trim()) { // D=Status(4), E=Tanggal Keluar(5), F=Alasan(6) sheet.getRange(i + 1, 4).setValue(data.status); sheet.getRange(i + 1, 5).setValue(data.tanggalKeluar || ''); sheet.getRange(i + 1, 6).setValue(data.alasan || ''); return ContentService.createTextOutput(JSON.stringify({status: 'success'})) .setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({status: 'error', message: 'Nama tidak ditemukan'})) .setMimeType(ContentService.MimeType.JSON); } // 2. TAMBAH ANGGOTA BARU (Sheet1) if (data.sheet === 'Sheet1') { const sheet = ss.getSheetByName('Sheet1'); // Urutan persis sesuai Sheet1: NAMA, TANGGAL DAFTAR, JABATAN, STATUS, TANGGAL KELUAR, ALASAN sheet.appendRow([ data.nama, data.tanggal, data.jabatan, data.status || 'Aktif', data.tanggalKeluar || '', data.alasan || '' ]); return ContentService.createTextOutput(JSON.stringify({status: 'success'})) .setMimeType(ContentService.MimeType.JSON); } // 3. TAMBAH TRANSAKSI (Sheet2) if (data.sheet === 'Sheet2') { const sheet = ss.getSheetByName('Sheet2'); sheet.appendRow([ data.timestamp, data.transaksi, data.kode, data.caraBayar, data.nominal, data.keterangan ]); return ContentService.createTextOutput(JSON.stringify({status: 'success'})) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({status: 'error', message: 'Perintah tidak dikenali'})) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({status: 'error', message: error.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { return ContentService.createTextOutput('KGP API Active - ' + new Date().toISOString()) .setMimeType(ContentService.MimeType.TEXT); }
